import { useEffect, useRef } from "react";
import Lenis from "lenis";
import gsap from "gsap";
import { ScrollTrigger } from "gsap/ScrollTrigger";

gsap.registerPlugin(ScrollTrigger);

export default function useLenis(isLoading) {
  const lenisRef = useRef(null);

  useEffect(() => {
    const lenis = new Lenis({
      duration: 1.2,
      easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)),
      smoothWheel: true,
    });

    lenisRef.current = lenis;
    lenis.stop(); // start paused, resume once loading finishes

    lenis.on("scroll", ScrollTrigger.update);

    const raf = (time) => lenis.raf(time * 1000);
    gsap.ticker.add(raf);
    gsap.ticker.lagSmoothing(0);

    return () => {
      lenis.destroy();
      gsap.ticker.remove(raf);
    };
  }, []);

  useEffect(() => {
    if (!lenisRef.current) return;
    if (isLoading) {
      lenisRef.current.stop();
    } else {
      lenisRef.current.start();
    }
  }, [isLoading]);
}

import { useEffect, useRef } from "react";

const clamp01 = (x) => Math.min(1, Math.max(0, x));

// Calls onProgress(p, rect) whenever the element moves.
// p = 0 → 1 across the element's sticky travel (height − viewport).
// Writes straight to the DOM from the callback, so there are no React re-renders.
export default function useScrollProgress(ref, onProgress) {
  const cb = useRef(onProgress);
  cb.current = onProgress;

  useEffect(() => {
    let raf;
    let lastTop = NaN;

    const reset = () => (lastTop = NaN);
    window.addEventListener("resize", reset);

    const tick = () => {
      const el = ref.current;
      if (el) {
        const r = el.getBoundingClientRect();
        if (r.top !== lastTop) {
          lastTop = r.top;
          const total = r.height - window.innerHeight;
          const p = total > 0 ? clamp01(-r.top / total) : 0;
          cb.current(p, r);
        }
      }
      raf = requestAnimationFrame(tick);
    };
    raf = requestAnimationFrame(tick);

    return () => {
      cancelAnimationFrame(raf);
      window.removeEventListener("resize", reset);
    };
  }, [ref]);
}

import { useEffect, useRef, useState } from "react";
import gsap from "gsap";
import useStore from "../../store/useStore";
import FerrariLogo from "../ui/FerrariLogo";

const TOP_SPEED = 340; // km/h — real Ferrari top speed, authentic detail

function Preloader() {
  const wrapperRef = useRef(null);
  const svgHolderRef = useRef(null);
  const counterRef = useRef(null);
  const [mounted, setMounted] = useState(true);
  const setIsLoading = useStore((state) => state.setIsLoading);

  useEffect(() => {
    const svg = svgHolderRef.current;
    const mainPaths = svg.querySelectorAll(".ferrari-main");
    const detailPaths = svg.querySelectorAll(".ferrari-detail");
    const highlightPaths = svg.querySelectorAll(".ferrari-highlight");
    const drawPaths = [...mainPaths, ...detailPaths];

    // Prep main + detail paths for stroke draw-in
    drawPaths.forEach((path) => {
      const length = path.getTotalLength();
      gsap.set(path, {
        strokeDasharray: length,
        strokeDashoffset: length,
        stroke: "#ff0000",
        strokeWidth: 0.6,
        fill: "rgba(255,0,0,0)",
      });
    });

    // Highlights start invisible — they "shine in" after the fill
    gsap.set(highlightPaths, { opacity: 0, fill: "#ffffff" });

    const progress = { value: 0 };

    const tl = gsap.timeline({
      onComplete: () => {
        setIsLoading(false);
        setMounted(false);
      },
    });

    // Draw outline, synced to km/h counter
    tl.to(progress, {
      value: TOP_SPEED,
      duration: 2.8,
      ease: "power2.inOut",
      onUpdate: () => {
        const pct = progress.value / TOP_SPEED;

        drawPaths.forEach((path) => {
          const length = path.getTotalLength();
          gsap.set(path, {
            strokeDashoffset: length - length * pct,
          });
        });

        if (counterRef.current) {
          counterRef.current.textContent = Math.floor(progress.value)
            .toString()
            .padStart(3, "0");
        }
      },
    });

    // Fill the main silhouette solid red
    tl.to(mainPaths, {
      fill: "rgba(255,0,0,1)",
      duration: 0.4,
      ease: "power2.out",
    });

    // Shine details fade in white, on top of the red fill
    tl.to(
      highlightPaths,
      {
        opacity: 1,
        duration: 0.5,
        ease: "power2.out",
        stagger: 0.005,
      },
      "-=0.15"
    );

    // Flash effect
    tl.to(wrapperRef.current, {
      opacity: 0.15,
      duration: 0.07,
      yoyo: true,
      repeat: 3,
      ease: "power1.inOut",
    });

    // Explode / scale out to reveal Hero underneath
    tl.to(wrapperRef.current, {
      scale: 8,
      opacity: 0,
      filter: "blur(12px)",
      duration: 0.8,
      ease: "power4.in",
    });

    return () => tl.kill();
  }, [setIsLoading]);

  if (!mounted) return null;

  return (
    <div
      ref={wrapperRef}
      className="fixed inset-0 z-50 bg-black flex flex-col items-center justify-center gap-6"
    >
      <div ref={svgHolderRef}>
        <FerrariLogo className="w-40 h-40 md:w-56 md:h-56" />
      </div>

      <div className="flex items-center gap-2 font-mono text-white text-sm tracking-[0.3em]">
        <span ref={counterRef}>000</span>
        <span className="text-red-500">km/h</span>
      </div>
    </div>
  );
}

export default Preloader;

import { Suspense, useCallback, useEffect, useRef, useState } from "react";
import Scene from "../canvas/Scene";

import HeroText from "../canvas/HeroText";
import HeroOverlay from "../HeroOverlay";
import HeroTelemetry from "../HeroTelemetry";
import ErrorBoundary from "../ui/ErrorBoundary";
import { CARS, CAR_MODEL } from "../../data/cars";
import "../hero.css";

function Hero() {
  const [index, setIndex] = useState(0);
  const [active, setActive] = useState(true);
  const rootRef = useRef(null);

  const select = useCallback(
    (i) => setIndex((i + CARS.length) % CARS.length),
    [],
  );

  // render the 3D scene only while the hero is (nearly) on screen
  useEffect(() => {
    const el = rootRef.current;
    if (!el) return;
    const io = new IntersectionObserver(
      ([entry]) => setActive(entry.isIntersecting),
      { rootMargin: "200px 0px 200px 0px" },
    );
    io.observe(el);
    return () => io.disconnect();
  }, []);

  useEffect(() => {
    const onKey = (e) => {
      if (e.key === "ArrowRight") select(index + 1);
      if (e.key === "ArrowLeft") select(index - 1);
    };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [index, select]);

  const car = CARS[index];
  const theme = {
    "--c1": car.theme[0],
    "--c2": car.theme[1],
    "--c3": car.theme[2],
  };

  return (
    <div className="hero-root" id="top" ref={rootRef}>
      <section className="hero-stage" style={theme}>
        <div className="hero-canvas">
          <ErrorBoundary>
            <Suspense fallback={null}>
              <Scene active={active}>
                <CarModel url={CAR_MODEL} paint={car.paint} />
                <GroundPlane position={[-0.171, 0.079, 1.775]} />
                <HeroText position={[-8, 0.082, 2.6]} />
              </Scene>
            </Suspense>
          </ErrorBoundary>
        </div>

        <HeroOverlay
          car={car}
          index={index}
          total={CARS.length}
          pending={false}
          onSelect={select}

import { GLTFLoader } from "three/examples/jsm/loaders/GLTFLoader.js";
import { DRACOLoader } from "three/examples/jsm/loaders/DRACOLoader.js";

let loader = null;
function getLoader() {
  if (!loader) {
    loader = new GLTFLoader();
    const draco = new DRACOLoader();
    draco.setDecoderPath("/draco/"); // served from public/draco, no CDN
    draco.preload();
    loader.setDRACOLoader(draco);
  }
  return loader;
}

const cache = new Map();

// One download per URL, cached for the session.
export function loadGLB(url) {
  if (!cache.has(url)) {
    const p = new Promise((resolve, reject) => {
      getLoader().load(url, resolve, undefined, reject);
    });
    p.catch(() => cache.delete(url));
    cache.set(url, p);
  }
  return cache.get(url);
}
